cmake_minimum_required(VERSION 3.22)
project(kas3 CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON) # The libraries also go into the Godot extension, a shared library.
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

option(KAS3_SANITIZE "Build everything with address and undefined-behaviour sanitizers" OFF)
if(KAS3_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_compile_definitions(ASAN_ENABLED UBSAN_ENABLED) # godot-cpp refuses to build under them unless told.
    add_link_options(-fsanitize=address,undefined)
endif()

option(KAS3_WEB_THREADS "Build the Web Godot extension with pthreads" OFF)
option(KAS3_RUNTIME_ONLY "Build the Godot extension without the AS3 compiler front end (loads precompiled .kas3 modules)" OFF)
if(EMSCRIPTEN)
    # Every static library linked into a Web GDExtension side module must match godot-cpp's side-module ABI,
    # and hide internal symbols so only GDE_EXPORT entry points are resolved dynamically against Godot.
    add_compile_options(-sSIDE_MODULE=1 -sSUPPORT_LONGJMP=wasm -fvisibility=hidden)
    if(CMAKE_BUILD_TYPE STREQUAL "Release" OR CMAKE_BUILD_TYPE STREQUAL "MinSizeRel")
        add_compile_options(-Oz)
    endif()
    if(KAS3_WEB_THREADS)
        add_compile_options(-sUSE_PTHREADS=1)
    endif()
endif()

# The front end must be loadable into the Godot editor, which is built without exceptions or RTTI.
file(GLOB KAS3_SOURCES CONFIGURE_DEPENDS src/*.cpp)

# The built-in declarations (builtins/*.as) are embedded as string literals.
file(GLOB KAS3_BUILTINS CONFIGURE_DEPENDS builtins/*.as)
list(SORT KAS3_BUILTINS)
set(builtins_cpp "// Generated by CMake from compiler/builtins. Do not edit.\n#include \"builtins.h\"\n\nnamespace kas3 {\n\nconst BuiltinFile BUILTIN_FILES[] = {\n")
foreach(builtin ${KAS3_BUILTINS})
    get_filename_component(builtin_name ${builtin} NAME)
    file(READ ${builtin} builtin_text)
    string(APPEND builtins_cpp "\t{ \"<built-in>/${builtin_name}\", R\"KAS3SRC(${builtin_text})KAS3SRC\" },\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${builtin})
endforeach()
string(APPEND builtins_cpp "};\nconst int BUILTIN_FILE_COUNT = sizeof(BUILTIN_FILES) / sizeof(BUILTIN_FILES[0]);\n\n} // namespace kas3\n")
set(builtins_path "${CMAKE_CURRENT_BINARY_DIR}/builtins_data.cpp")
set(builtins_old "")
if(EXISTS ${builtins_path})
    file(READ ${builtins_path} builtins_old)
endif()
if(NOT "${builtins_old}" STREQUAL "${builtins_cpp}")
    file(WRITE ${builtins_path} "${builtins_cpp}")
endif()

# Godot's API as AS3 declarations (godot/godot.as, written by godot/generate.py) is embedded too.
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(godot_path "${CMAKE_CURRENT_BINARY_DIR}/godot_data.cpp")
add_custom_command(OUTPUT ${godot_path}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/godot/embed.py
        ${CMAKE_CURRENT_SOURCE_DIR}/godot/godot.as ${godot_path}
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/godot/godot.as ${CMAKE_CURRENT_SOURCE_DIR}/godot/embed.py
    COMMENT "Embedding Godot's API declarations")

# So are Godot's own names for what godot.as declares (godot/godot_names.txt, by the same generator).
set(godot_names_path "${CMAKE_CURRENT_BINARY_DIR}/godot_names_data.cpp")
add_custom_command(OUTPUT ${godot_names_path}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/godot/embed.py
        ${CMAKE_CURRENT_SOURCE_DIR}/godot/godot_names.txt ${godot_names_path} godot_names_text
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/godot/godot_names.txt ${CMAKE_CURRENT_SOURCE_DIR}/godot/embed.py
    COMMENT "Embedding Godot's names")

add_library(kas3front STATIC ${KAS3_SOURCES} ${builtins_path} ${godot_path} ${godot_names_path})
target_include_directories(kas3front PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(kas3front PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# The typed IR data model and binary codec depend on nothing in src/, so a runtime-only extension links
# without the front end or Godot's embedded declaration tables.
file(GLOB KAS3_IR_CORE_SOURCES CONFIGURE_DEPENDS ir/ir_*.cpp)
add_library(kas3ir_core STATIC ${KAS3_IR_CORE_SOURCES})
target_include_directories(kas3ir_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/ir)
target_compile_options(kas3ir_core PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# The lowerer turns the checked syntax tree into typed IR.
file(GLOB KAS3_IR_LOWER_SOURCES CONFIGURE_DEPENDS ir/lower*.cpp)
add_library(kas3ir STATIC ${KAS3_IR_LOWER_SOURCES})
target_include_directories(kas3ir PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/ir)
target_link_libraries(kas3ir PUBLIC kas3ir_core kas3front)
target_compile_options(kas3ir PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# The interpreter and its runtime. No Godot header: the engine is reached through vm/host.h.
file(GLOB KAS3_VM_SOURCES CONFIGURE_DEPENDS vm/*.cpp)
add_library(kas3vm STATIC ${KAS3_VM_SOURCES})
target_include_directories(kas3vm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/vm)
target_link_libraries(kas3vm PUBLIC kas3ir_core)
target_compile_options(kas3vm PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# The C++ backend: the IR written as C++ source. It includes ir/ only, and writes text.
file(GLOB KAS3_BACKEND_CPP_SOURCES CONFIGURE_DEPENDS backend_cpp/*.cpp)
add_library(kas3backend_cpp STATIC ${KAS3_BACKEND_CPP_SOURCES})
target_include_directories(kas3backend_cpp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/backend_cpp)
target_link_libraries(kas3backend_cpp PUBLIC kas3ir_core)
target_compile_options(kas3backend_cpp PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# What generated code is built against and, where there is no engine, the host it runs in.
file(GLOB KAS3_RUNTIME_SOURCES CONFIGURE_DEPENDS runtime/*.cpp)
# `main` for a program library built as a plain program is a library of its own: the Godot extension takes
# the whole runtime in, and has no use for one.
list(FILTER KAS3_RUNTIME_SOURCES EXCLUDE REGEX "standalone_main\\.cpp$")
add_library(kas3standalone STATIC runtime/standalone_main.cpp)
target_include_directories(kas3standalone PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/runtime ${CMAKE_CURRENT_SOURCE_DIR}/vm ${CMAKE_CURRENT_SOURCE_DIR}/ir)
target_compile_options(kas3standalone PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)
add_library(kas3runtime STATIC ${KAS3_RUNTIME_SOURCES})
target_include_directories(kas3runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/runtime)
target_link_libraries(kas3runtime PUBLIC kas3vm)
target_compile_options(kas3runtime PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# Tests link just this object ahead of the shipping archive to enable native counters.
add_library(kas3_counted_host OBJECT runtime/host.cpp)
target_link_libraries(kas3_counted_host PRIVATE kas3runtime)
target_compile_options(kas3_counted_host PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)
target_compile_definitions(kas3_counted_host PRIVATE KAS3_TEST_COUNTS)

# The language server: JSON, the workspace and the message loop. The Godot editor does not need it.
file(GLOB KAS3_LSP_SOURCES CONFIGURE_DEPENDS lsp/*.cpp)
add_library(kas3lsp STATIC ${KAS3_LSP_SOURCES})
target_include_directories(kas3lsp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/lsp)
target_link_libraries(kas3lsp PUBLIC kas3front)
target_compile_options(kas3lsp PRIVATE -fno-exceptions -fno-rtti -Wall -Wextra -Werror)

# The Godot extension: AS3 as a Godot scripting language. Optional, because it needs godot-cpp, which is not
# in this repository:  cmake -DKAS3_GODOT_CPP=<path to a godot-cpp checkout for Godot 4.7> ...
set(KAS3_GODOT_CPP "" CACHE PATH "A godot-cpp checkout; empty to leave the Godot extension out")
if(KAS3_GODOT_CPP)
    set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
    set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
    if(EMSCRIPTEN)
        set(GODOTCPP_THREADS ${KAS3_WEB_THREADS} CACHE BOOL "" FORCE)
    endif()
    add_subdirectory(${KAS3_GODOT_CPP} ${CMAKE_CURRENT_BINARY_DIR}/godot-cpp EXCLUDE_FROM_ALL)
    file(GLOB KAS3_GODOT_SOURCES CONFIGURE_DEPENDS godot_ext/*.cpp)
    set(KAS3_GODOT_EXTRA_SOURCES "")
    set(KAS3_HAS_PLAY_BACKEND OFF)
    if(NOT EMSCRIPTEN AND NOT KAS3_RUNTIME_ONLY AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
        set(KAS3_HAS_PLAY_BACKEND ON)
        # What Play builds a project's program library from is carried by the extension: the headers generated
        # code is built against, and one word for them and the backend together (tools/embed_runtime.py).
        file(GLOB KAS3_PLAY_INPUTS CONFIGURE_DEPENDS runtime/*.h vm/*.h ir/*.h backend_cpp/*.cpp backend_cpp/*.h)
        set(play_runtime_path "${CMAKE_CURRENT_BINARY_DIR}/play_runtime_data.cpp")
        add_custom_command(OUTPUT ${play_runtime_path}
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_runtime.py ${CMAKE_CURRENT_SOURCE_DIR} ${play_runtime_path}
            DEPENDS ${KAS3_PLAY_INPUTS} ${CMAKE_CURRENT_SOURCE_DIR}/tools/embed_runtime.py
            COMMENT "Embedding the runtime's headers")
        set(KAS3_GODOT_EXTRA_SOURCES "${play_runtime_path}")
    endif()
    set(runtime_names "${CMAKE_CURRENT_BINARY_DIR}/runtime_names.gen.h")
    add_custom_command(OUTPUT ${runtime_names}
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/godot/runtime_names.py
            ${CMAKE_CURRENT_SOURCE_DIR}/godot/godot_names.txt ${runtime_names}
        DEPENDS godot/runtime_names.py godot/godot_names.txt)
    set(export_sources godot_ext/godot_named.cpp godot_ext/program.cpp godot_ext/native_program.cpp godot_ext/godot_utility.cpp)
    foreach(source IN LISTS export_sources)
        list(REMOVE_ITEM KAS3_GODOT_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
    endforeach()
    add_library(kas3_godot_common OBJECT ${KAS3_GODOT_SOURCES} ${KAS3_GODOT_EXTRA_SOURCES} ${runtime_names})
    get_target_property(KAS3_GODOT_CPP_INCLUDES godot-cpp INTERFACE_INCLUDE_DIRECTORIES)
    if(KAS3_GODOT_CPP_INCLUDES)
        target_include_directories(kas3_godot_common SYSTEM PUBLIC ${KAS3_GODOT_CPP_INCLUDES})
    endif()
    target_include_directories(kas3_godot_common PUBLIC ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/godot_ext)
    if(NOT KAS3_RUNTIME_ONLY)
        target_include_directories(kas3_godot_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
    endif()
    # godot-cpp's casts need run-time type information, so this glue has it; the interpreter does not.
    target_compile_options(kas3_godot_common PUBLIC -fno-exceptions -Wall -Wextra -Werror)
    if(KAS3_RUNTIME_ONLY)
        target_compile_definitions(kas3_godot_common PUBLIC KAS3_RUNTIME_ONLY=1)
        target_link_libraries(kas3_godot_common PUBLIC godot-cpp kas3runtime kas3vm kas3ir_core)
    elseif(KAS3_HAS_PLAY_BACKEND)
        target_link_libraries(kas3_godot_common PUBLIC godot-cpp kas3runtime kas3vm kas3ir kas3backend_cpp)
    else()
        target_link_libraries(kas3_godot_common PUBLIC godot-cpp kas3runtime kas3vm kas3ir)
    endif()
    add_library(kas3_godot SHARED ${export_sources})
    target_link_libraries(kas3_godot PRIVATE kas3_godot_common)
    # On this machine's own system the library goes beside the test project, which loads it. Built for
    # another (iOS), it stays in the build folder: tests/build_ios.sh makes an .xcframework of it.
    # -DKAS3_GODOT_OUTPUT=<folder> puts it elsewhere: a build under the sanitizers must not take the place of
    # the one the test project loads (tests/README.md).
    if(KAS3_GODOT_OUTPUT)
    elseif(CMAKE_CROSSCOMPILING)
        set(KAS3_GODOT_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/out)
    else()
        set(KAS3_GODOT_OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/tests/godot_project/bin)
    endif()
    set_target_properties(kas3_godot PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY ${KAS3_GODOT_OUTPUT}
        OUTPUT_NAME kas3_godot)
    if(KAS3_HAS_PLAY_BACKEND)
        add_library(kas3_godot_counts SHARED vm/runtime.cpp vm/interpreter_named_site.cpp tests/script_read_allocations.cpp ${export_sources} $<TARGET_OBJECTS:kas3_counted_host>)
        target_link_libraries(kas3_godot_counts PRIVATE kas3_godot_common)
        target_compile_definitions(kas3_godot_counts PRIVATE KAS3_TEST_COUNTS)
        set_target_properties(kas3_godot_counts PROPERTIES OUTPUT_NAME kas3_godot
            LIBRARY_OUTPUT_DIRECTORY "${KAS3_GODOT_OUTPUT}/test_counts")
    endif()
    if(EMSCRIPTEN)
        set(KAS3_WEB_EXPORTS -Wl,--export=kas3_godot_init)
        if(NOT KAS3_RUNTIME_ONLY)
            list(APPEND KAS3_WEB_EXPORTS
                -Wl,--export=kas3_web_reload
                -Wl,--export=kas3_web_last_error
                -Wl,--export=kas3_web_reset_sources)
        endif()
        target_link_options(kas3_godot PRIVATE -sSIDE_MODULE=2 ${KAS3_WEB_EXPORTS})
        if(CMAKE_BUILD_TYPE STREQUAL "Release" OR CMAKE_BUILD_TYPE STREQUAL "MinSizeRel")
            target_link_options(kas3_godot PRIVATE -Oz)
        endif()
        set_target_properties(kas3_godot PROPERTIES SUFFIX ".wasm")
    endif()
endif()
